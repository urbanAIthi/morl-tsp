\sisetup{round-mode=places,round-precision=2}
\newcommand{\heat}[1]{%
  \pgfmathsetmacro{\absrho}{abs(#1)}%
\pgfmathsetmacro{\pct}{ifthenelse(\absrho<0.45,0,min(100,30*\absrho))}% white below 0.45
  \ifdim #1 pt > 0pt
    \edef\heatcolor{blue!\pct!white}%
  \else
    \edef\heatcolor{red!\pct!white}%
  \fi
  \expandafter\cellcolor\expandafter{\heatcolor}\num{#1}%
}
